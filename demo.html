Usage Example

import { QArt } from 'qart'; const qr = new QArt({ data: 'https://example.com', width: 300, height: 300, dotsOptions: { type: 'rounded', gradient: { type: 'linear', rotation: Math.PI / 4, colorStops: [ { offset: 0, color: '#667eea' }, { offset: 1, color: '#764ba2' } ] } }, backgroundOptions: { color: '#1a1a2e' }, cornersSquareOptions: { type: 'rounded', color: '#667eea' } }); qr.append(document.getElementById('container')); await qr.download({ name: 'my-qr', extension: 'png' });
